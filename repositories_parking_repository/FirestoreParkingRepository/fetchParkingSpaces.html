@override
Future<List<ParkingSpace>> fetchParkingSpaces(String destinationKey) async {
  final snapshot = await _parkingCollection(destinationKey).get();

  return snapshot.docs
      .map((doc) => ParkingSpace.fromMap(doc.id, doc.data()))
      .toList();
}
